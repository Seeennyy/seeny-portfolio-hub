                let teammatesHtml = '';
                if (project.teammates && project.teammates.length > 0) {
                    teammatesHtml = `<h6 class="font-semibold text-teal-700 uppercase tracking-wide text-xs mb-1 mt-3">Teammates</h6><p class="text-gray-600 text-sm">`;
                    project.teammates.forEach((tm, index) => {
                        if (tm.linkedin) {
                            teammatesHtml += `<a href="${tm.linkedin}" target="_blank" class="text-teal-600 hover:underline">${tm.name}</a>`;
                        } else {
                            teammatesHtml += tm.name;
                        }
                        if (index < project.teammates.length - 1) {
                            teammatesHtml += ', ';
                        }
                    });
                    teammatesHtml += `</p>`;
                }

                let documentationLinkHtml = '';
                if (project.documentationLink) {
                    documentationLinkHtml = `<a href="${project.documentationLink}" target="_blank" class="text-teal-600 hover:underline text-sm font-semibold mt-3 block">View Documentation</a>`;
                }

                projectCard.innerHTML = `
                    <div>
                        <h5 class="text-lg md:text-xl font-bold text-gray-800 mb-2">${project.title}</h5>
                        <p class="text-gray-600 text-sm mb-4">${project.summary}</p>
                        <div id="details-${project.id}" class="project-details-content">
                            <h6 class="font-semibold text-teal-700 uppercase tracking-wide text-xs mb-1">The Challenge</h6>
                            <p class="text-gray-600 text-sm mb-3">${project.challenge}</p>
                            <h6 class="font-semibold text-teal-700 uppercase tracking-wide text-xs mb-1">The Project</h6>
                            <p class="text-gray-600 text-sm mb-3">${project.project}</p>
                            <h6 class="font-semibold text-teal-700 uppercase tracking-wide text-xs mb-1">The Impact</h6>
                            <p class="text-gray-600 text-sm">${project.impact}</p>
                            ${teammatesHtml}
                            ${documentationLinkHtml}
                        </div>
                    </div>
                    <button class="view-details-btn mt-4 bg-teal-500 text-white px-4 py-2 rounded-lg hover:bg-teal-600 transition duration-300 text-sm font-semibold self-start" data-project-id="${project.id}">View Details</button>
                `;
                projectsContainer.appendChild(projectCard);
            });

            document.querySelectorAll('.view-details-btn').forEach(button => {
                button.addEventListener('click', (event) => {
                    const projectId = event.target.dataset.projectId;
                    const detailsDiv = document.getElementById(`details-${projectId}`);
                    detailsDiv.classList.toggle('expanded');
                    if (detailsDiv.classList.contains('expanded')) {
                        event.target.textContent = 'Hide Details';
                    } else {
                        event.target.textContent = 'View Details';
                    }
                });
            });
        }
